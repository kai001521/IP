B. Gaussian Low Pass Filter

08

Aim: To smooth an image using a Gaussian low-pass filter.

clc;
clear;

S = imread(fullpath(getIPCVpath()+"/images/lena.png"));
if size(S,3)==3 then
    S = rgb2gray(S);
end

S = double(S)/255;
[M,N] = size(S);

F = fftshift(fft2(S));
[X,Y] = meshgrid(1:N,1:M);

D = sqrt((X-N/2).^2+(Y-M/2).^2);
H = exp(-(D.^2)/(2*30^2));

R = real(fft(ifftshift(F.*H),1));

subplot(1,2,1);
imshow(S);
title('Original');

subplot(1,2,2);
imshow(R);
title('Gaussian Smoothed');

C. Max-Min and Min-Max Filters (object)

19

Aim: To apply max-min and min-max filters to an input image.

clc;
clear;
close;
img = imread("D:\gray.jpeg");
img = double(img);
[m, n] = size(img);
max_filtered = img;
min_filtered = img;
maxmin_filtered = img;
for i = 2:m-1
  for j = 2:n-1
      window = img(i-1:i+1, j-1:j+1);
      max_filtered(i,j) = max(window);
      min_filtered(i,j) = min(window);
      maxmin_filtered(i,j) = (max(window) + min(window)) / 2;
  end
end
max_filtered = uint8(max_filtered);
min_filtered = uint8(min_filtered);
maxmin_filtered = uint8(maxmin_filtered);
figure();
subplot(2,2,1);
imshow(uint8(img));
title("Original Image");
subplot(2,2,2);
imshow(max_filtered);
title("Max Filter");
subplot(2,2,3);
imshow(min_filtered);
title("Min Filter");
subplot(2,2,4);
imshow(maxmin_filtered);
title("Max-Min Filter");
